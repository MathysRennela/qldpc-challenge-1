\section{Quantum error correction}
% =====================================================================

\begin{frame}{Why error correction at all}
  \begin{itemize}\setlength{\itemsep}{8pt}
    \item Physical qubits err at rates $\sim 10^{-3}$--$10^{-2}$ per operation;
          useful algorithms need $10^{-10}$ or better.
    \item Fix: encode $k$ \textbf{logical} qubits redundantly into $n$
          \textbf{physical} qubits, measure carefully chosen parities
          (\emph{syndromes}), and undo errors faster than they accumulate.
    \item \textbf{Threshold theorem}: below a constant physical error rate, logical
          errors can be suppressed exponentially in the code distance $d$.
    \item The whole game is the \textbf{overhead}: how many physical qubits per
          logical qubit, at a given protection level?
  \end{itemize}
\end{frame}

\begin{frame}{Stabilizer codes and the CSS recipe}
  A CSS code is specified by two binary parity-check matrices:
  \[
    H_X \in \GF^{\,r_X \times n}, \qquad
    H_Z \in \GF^{\,r_Z \times n}, \qquad
    \text{with } H_X H_Z^{\mathsf T} = 0 \ (\mathrm{mod}\ 2).
  \]
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \begin{itemize}\setlength{\itemsep}{6pt}
        \item Each row of $H_X$ is an $X$-type parity check (a stabilizer);
              each row of $H_Z$ a $Z$-type one. The orthogonality condition makes
              them all commute.
        \item Number of logical qubits:
              \[ k = n - \rank H_X - \rank H_Z. \]
        \item Everything about the code is \emph{recomputable} from
              $(H_X, H_Z)$ --- a fact the Challenge leans on heavily.
      \end{itemize}
    \end{column}
    \begin{column}{0.4\textwidth}
      \begin{block}{Errors, split in two}
        Bit flips ($X$) are caught by $Z$-checks; phase flips ($Z$) by
        $X$-checks. A CSS code is two classical codes stitched together at the
        commutation condition.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{The benchmark: the surface code}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \begin{itemize}\setlength{\itemsep}{8pt}
        \item Qubits on a 2D grid; every check touches only its
              \textbf{4 neighbours} $\Rightarrow$ easy to build and measure.
        \item $d \sim \sqrt{n}$, but \textbf{one} logical qubit per patch:
              $k/n \to 0$ --- a useful machine needs millions of physical
              qubits.
        \item Already LDPC; the problem is \textbf{2D locality},
              not check weight.
      \end{itemize}
      \medskip
      \emph{Can we keep the low-weight checks, drop the locality, and get a
      constant rate?}
    \end{column}
    \begin{column}{0.42\textwidth}
      \centering
      \begin{tikzpicture}[scale=0.52]
        \foreach \i in {0,...,5} \foreach \j in {0,...,5}
          \fill[muted] (\i,\j) circle (2.2pt);

      \end{tikzpicture}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Quantum LDPC codes, and the yardstick}
  \textbf{LDPC} $=$ low-density parity checks: every check touches $O(1)$ qubits
  and every qubit sits in $O(1)$ checks --- the property that makes syndrome
  measurement practical.
  \begin{itemize}\setlength{\itemsep}{6pt}
    \item Breakthrough (2020s): families with \textbf{constant rate and linear
          distance} exist. Practical frontier today: e.g.\ IBM's
          bivariate-bicycle ``gross code'' $[[144,12,12]]$ --- 12 logical
          qubits at distance 12 from 144 data qubits, weight-6 checks.
    \item Headline figure of merit:
          \[ \text{efficiency} \;=\; \frac{k\,d^2}{n}
             \qquad (\text{surface code} \approx 1,\quad
                     \text{gross code} = 12,\quad
                     \text{board record} = 23.81). \]
  \end{itemize}
  \begin{block}{Why $kd^2/n$? (Bravyi--Poulin--Terhal, 2010)}
    Any 2D-\emph{local} code obeys $kd^2 \le c\,n$; the surface code saturates
    it at $\approx 1$. So $kd^2/n \gg 1$ \textbf{quantifies how far beyond a
    strict 2D layout} a code performs --- the record of $23.81$
    ($[[336,20,20]]$, a two-block group-algebra code on $\mathrm{PSL}(2,7)$)
    is a statement about connectivity, not just cleverness.
  \end{block}
\end{frame}

\section{Distance}
% =====================================================================

\begin{frame}{Distance: the hard part}
  A \textbf{logical operator} acts on the encoded qubits without triggering any
  check; $d$ is the weight of the cheapest one --- the cheapest
  \emph{undetectable error} --- and it enters the figure of merit
  \emph{squared}.
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \begin{itemize}\setlength{\itemsep}{6pt}
        \item \alert{Computing $d$ exactly is NP-hard.} Every distance in every
              paper came from a bounded search --- and bounded searches miss
              things.
        \item But any search that \emph{finds} a logical yields an explicit
              \textbf{witness}: a vector anyone can check in milliseconds.
              It proves $d \le \wt(v)$ --- trustlessly.
        \item Exact values ($d = v$): integer program per logical class ---
              feasible for small codes, hopeless for large high-rate ones.
      \end{itemize}
    \end{column}
    \begin{column}{0.44\textwidth}
      \begin{alertblock}{The trap}
        A \emph{high} estimate from a \emph{short} search looks like a great
        code. It usually means the search hasn't found the light logical yet.
        Distance claims need adversarial checking.
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{The two corroboration searches}
  Both only ever \emph{exhibit} a logical of some weight --- each find tightens
  $d \le w$, neither proves a lower bound. They attack the claim by
  \textbf{independent mechanisms}:
  \smallskip
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{Random information set (RIS)} \hfill {\color{muted}\scriptsize algebraic}
      \begin{itemize}\setlength{\itemsep}{4pt}\small
        \item Basis of $\ker H_Z$: everything that commutes with the checks.
        \item Each trial: shuffle the columns, RREF the basis --- random
              re-coordinatizations make different vectors \emph{sparse}.
        \item Light rows (and pairwise sums) that are not stabilizer products
              are logical operators; keep the lightest.
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{BP+OSD decoder residuals} \hfill {\color{muted}\scriptsize operational}
      \begin{itemize}\setlength{\itemsep}{4pt}\small
        \item Each trial: inject a random weight-$w$ error $e$, hand its
              syndrome to a BP+OSD decoder, get correction $\hat e$.
        \item The residual $e + \hat e$ has zero syndrome by construction;
              if it acts on the logicals, it \emph{is} a logical operator.
        \item A good decoder's failures concentrate on the \emph{cheapest}
              undetectable errors --- exactly what $d$ measures.
      \end{itemize}
    \end{column}
  \end{columns}
  \medskip
  \begin{block}{}
    \small Survives \textbf{both} at deep budgets $\Rightarrow$ corroborated
    ($d \le v^{*}$); either finds lighter $\Rightarrow$
    \textcolor{flag}{refuted}, witness attached.
  \end{block}
\end{frame}

\section{The Challenge}
% =====================================================================

\begin{frame}{What the Challenge is}
  \begin{itemize}\setlength{\itemsep}{8pt}
    \item A \textbf{public leaderboard} of qLDPC codes: one JSON file per code,
          in a git repository, with a static site on top.
          \hfill {\color{muted}\small 49 codes today}
    \item Every claim is \textbf{recomputed from the parity checks} --- $n$, $k$,
          CSS commutation, check weights, distance witnesses, 2D-locality radius.
          Nothing is trusted from the submitter.
    \item Codes compete on \textbf{Pareto frontiers} inside track cells
          (check-weight class $\times$ locality class), never on a single score.
    \item Designed so that \textbf{LLM research agents} can do the searching ---
          with a referee they cannot talk their way past.
  \end{itemize}
\end{frame}

    No code is a find until \code{validate\_candidate} returns

\begin{frame}{The CI gauntlet, per submission}
  \small
  \begin{enumerate}\setlength{\itemsep}{5pt}
    \item \textbf{Scope}: a PR adding a code may not touch workflows, the schema,
          or \code{verify/} --- a submission cannot change its own judge.
          One new code per PR.
    \item \textbf{Integrity pin}: SHA-256 of the whole trusted stack against a
          committed manifest; any drift fails the build.
    \item \textbf{Self-tests}: the verifier must still \emph{reject} known-bad
          submissions and known over-claims.
    \item \textbf{Recomputation}: $n$, $k$, CSS, weights, witnesses --- recomputed
          by the \emph{base branch's} verifier. Locality is \textbf{measured}
          from a submitted layout certificate, never declared; no honest
          layout $\Rightarrow$ the code competes as \code{unrestricted}.
    \item \textbf{Adaptive refutation}: two independent searches attack the
          distance claim, budget scaling with size ($2500 + 40n$ trials).
          A claim that advances its cell's frontier triggers the deep tier:
          a $\sim$150$\times$-deeper accelerated search plus a Python canary
          seed (three full Python seeds if the accelerator is unavailable).
          One lighter logical found $\Rightarrow$ PR fails, witness printed.
    \item \textbf{Authorship}: the PR author must be a listed author of the code.
  \end{enumerate}
\end{frame}

    \item The other defense is time: a \textbf{weekly random-seed sweep}

  \end{itemize}
\end{frame}

\begin{frame}{What it is for}
  \begin{itemize}\setlength{\itemsep}{9pt}
    \item A \textbf{shared, machine-checked record} of the best qLDPC codes
          under each constraint --- parameters that today live scattered
          across papers.
    \item A \textbf{benchmark for AI research agents}: a tool-agnostic,
          pure-numpy starter kit plus an operating manual
          (\code{AUTORESEARCH.md}) any model can be pointed at.
    \item \textbf{Shared field notes}: campaigns end by distilling hard-won
          lessons into the manual --- every newcomer, human or agent, starts
          where the last one stopped.
    \item A template for \textbf{verification-first open science}: untrusted
          exploration, a pinned referee, claims that decay with evidence
          instead of persisting as folklore.
  \end{itemize}
